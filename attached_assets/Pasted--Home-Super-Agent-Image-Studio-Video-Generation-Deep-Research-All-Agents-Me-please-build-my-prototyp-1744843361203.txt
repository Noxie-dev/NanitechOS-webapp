    <script>
        // Update clock
        function updateClock() {
            const now = new Date();
            let hours = now.getHours();
            let minutes = now.getMinutes();
            
            // Format time
            hours = hours < 10 ? '0' + hours : hours;
            minutes = minutes < 10 ? '0' + minutes : minutes;
            
            document.getElementById('clock').textContent = hours + ':' + minutes;
        }
        
        // Update clock initially and set interval
        updateClock();
        setInterval(updateClock, 60000);
        
        // App Grid Toggle
        const appGridToggle = document.getElementById('app-grid-toggle');
        const appGrid = document.getElementById('app-grid');
        
        appGridToggle.addEventListener('click', () => {
            appGrid.classList.toggle('active');
        });
        
        // Close app grid when clicking elsewhere
        document.addEventListener('click', (e) => {
            if (!appGridToggle.contains(e.target) && !appGrid.contains(e.target) && appGrid.classList.contains('active')) {
                appGrid.classList.remove('active');
            }
        });
        
        // Settings Panel Toggle
        const settingsToggle = document.getElementById('settings-toggle');
        const quickSettings = document.getElementById('quick-settings-panel');
        
        settingsToggle.addEventListener('click', (e) => {
            e.stopPropagation();
            quickSettings.classList.toggle('active');
        });
        
        // Close settings panel when clicking elsewhere
        document.addEventListener('click', (e) => {
            if (!settingsToggle.contains(e.target) && !quickSettings.contains(e.target) && quickSettings.classList.contains('active')) {
                quickSettings.classList.remove('active');
            }
        });
        
        // Window Management
        const windows = document.querySelectorAll('.window');
        let activeZIndex = 10;
        
        // Make windows draggable
        windows.forEach(window => {
            const header = window.querySelector('.window-header');
            let isDragging = false;
            let offsetX, offsetY;
            
            header.addEventListener('mousedown', (e) => {
                isDragging = true;
                offsetX = e.clientX - window.getBoundingClientRect().left;
                offsetY = e.clientY - window.getBoundingClientRect().top;
                
                // Bring window to front
                window.style.zIndex = ++activeZIndex;
            });
            
            document.addEventListener('mousemove', (e) => {
                if (isDragging) {
                    const x = e.clientX - offsetX;
                    const y = e.clientY - offsetY;
                    
                    window.style.left = `${x}px`;
                    window.style.top = `${y}px`;
                }
            });
            
            document.addEventListener('mouseup', () => {
                isDragging = false;
            });
            
            // Close button functionality
            const closeBtn = window.querySelector('.window-close');
            closeBtn.addEventListener('click', () => {
                window.classList.remove('active');
            });
        });
        
        // Open windows from dock
        document.getElementById('files-icon').addEventListener('click', () => {
            const window = document.getElementById('files-window');
            window.classList.add('active');
            window.style.zIndex = ++activeZIndex;
        });
        
        document.getElementById('browser-icon').addEventListener('click', () => {
            const window = document.getElementById('browser-window');
            window.classList.add('active');
            window.style.zIndex = ++activeZIndex;
        });
        
        // Terminal typing animation
        const typingElements = document.querySelectorAll('.typing-text');
        
        typingElements.forEach(element => {
            const text = element.textContent;
            element.textContent = '';
            
            let i = 0;
            const typeChar = () => {
                if (i < text.length) {
                    element.textContent += text.charAt(i);
                    i++;
                    setTimeout(typeChar, Math.random() * 50 + 50);
                }
            };
            
            setTimeout(() => {
                typeChar();
            }, 1000);
        });
        
        // Open windows from app grid
        document.querySelectorAll('.app-item').forEach(item => {
            item.addEventListener('click', () => {
                // Close the app grid when an app is selected
                appGrid.classList.remove('active');
                
                // Check which app was clicked and open corresponding window
                const appName = item.querySelector('span').textContent;
                
                if (appName === 'Terminal') {
                    const window = document.getElementById('terminal-window');
                    window.classList.add('active');
                    window.style.zIndex = ++activeZIndex;
                } else if (appName === 'Files' || appName === 'NaniVault') {
                    const window = document.getElementById('files-window');
                    window.classList.add('active');
                    window.style.zIndex = ++activeZIndex;
                } else if (appName === 'Browser' || appName === 'Firefox' || appName === 'Brave Web Browser') {
                    const window = document.getElementById('browser-window');
                    window.classList.add('active');
                    window.style.zIndex = ++activeZIndex;
                } else if (appName === 'Settings') {
                    const window = document.getElementById('settings-window');
                    window.classList.add('active');
                    window.style.zIndex = ++activeZIndex;
                }
            });
        });
